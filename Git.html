Git · карточки для блокнота · вырезать по пунктиру ✂

⚙️ Настройка и старт

git config --global user.name "Имя"Установить имя
git config --global user.email "email"Установить email
git config --listПоказать все настройки
git config --global core.editor "code --wait"Редактор по умолчанию
git initНовый локальный репозиторий
git clone <url>Клонировать удалённый репозиторий
git remote add origin <url>Добавить удалённый репозиторий
git remote -vСписок удалённых репозиториев

📝 Ежедневные команды

git statusСтатус рабочей директории
git add <файл>Добавить файл в индекс
git add .Добавить все изменения
git add -pИнтерактивное добавление
git commit -m "сообщение"Зафиксировать изменения
git commit -am "сообщение"Добавить и зафиксировать (для отслеживаемых)
git commit --amendИсправить последний коммит
git diffРазница между рабочей директорией и индексом
git diff --stagedРазница между индексом и последним коммитом
git diff HEADВсе изменения (индексированные и нет)

📜 История

git logПолная история коммитов
git log --onelineКраткая история (одна строка на коммит)
git log --graph --all --onelineГрафическая история всех веток
git log -p <файл>История изменений файла с диффами
git log --follow <файл>История файла, включая переименования
git show <commit>Показать изменения в коммите
git blame <файл>Кто и когда менял каждую строку

🌿 Ветки

git branchСписок локальных веток
git branch -aСписок всех веток (локальных и удалённых)
git branch <имя>Создать ветку
git switch <ветка>Переключиться на ветку (новый способ)
git checkout <ветка>Переключиться на ветку (старый способ)
git switch -c <новая>Создать и переключиться
git checkout -b <новая>Создать и переключиться (старый способ)
git merge <ветка>Слить ветку в текущую
git branch -d <ветка>Удалить ветку (если слита)
git branch -D <ветка>Принудительно удалить ветку

☁️ Удалённые репозитории

git fetchПолучить изменения без слияния
git pullПолучить и слить изменения
git pushОтправить изменения
git push -u origin <ветка>Отправить и связать с удалённой
git push origin --delete <ветка>Удалить удалённую ветку
git remote set-url origin <новый_url>Изменить URL удалённого репозитория

↩️ Отмена изменений

git restore <файл>Отменить изменения в рабочей директории
git restore --staged <файл>Убрать файл из индекса
git reset <файл>Убрать файл из индекса (старый способ)
git reset --hardСбросить всё к последнему коммиту (ОПАСНО)
git reset HEAD^Отменить последний коммит, сохранив изменения
git cleanУдалить неотслеживаемые файлы
git revert <commit>Создать коммит, отменяющий указанный

📦 Stash (временное сохранение)

git stashСпрятать текущие изменения
git stash push -m "сообщение"Спрятать с описанием
git stash listСписок спрятанных изменений
git stash popПрименить и удалить последний stash
git stash applyПрименить, но оставить в списке
git stash dropУдалить последний stash
git stash clearОчистить все stash

🔀 Rebase и Cherry-pick

git rebase <ветка>Перебазировать текущую ветку на другую
git rebase -i HEAD~5Интерактивный rebase последних 5 коммитов
git rebase --continueПродолжить после разрешения конфликтов
git rebase --abortПрервать rebase
git cherry-pick <commit>Применить указанный коммит в текущую ветку
git reflogИстория всех действий (спасательный круг)

🏷️ Теги

git tagСписок тегов
git tag v1.0.0Создать легковесный тег
git tag -a v1.0.0 -m "Release"Создать аннотированный тег
git show v1.0.0Информация о теге
git push origin v1.0.0Отправить тег на удалённый
git push origin --tagsОтправить все теги
git tag -d v1.0.0Удалить локальный тег
git push origin --delete v1.0.0Удалить удалённый тег

📦 Submodules

git submodule add <url>Добавить подмодуль
git submodule initИнициализировать подмодули
git submodule updateОбновить подмодули
git submodule update --init --recursiveКлонировать с подмодулями
git submodule statusСтатус подмодулей

⚡ Полезные алиасы

git config --global alias.st statusgit st = status
git config --global alias.co checkoutgit co = checkout
git config --global alias.br branchgit br = branch
git config --global alias.ci commitgit ci = commit
git config --global alias.lg "log --oneline --graph --all"git lg = красивая история
git config --global alias.unstage "reset HEAD --"git unstage = убрать из индекса

🔍 Поиск

git grep "текст"Поиск по содержимому файлов
git grep -n "текст"Поиск с номерами строк
git log -S "текст"Коммиты, где текст был добавлен/удалён
git log -G "regex"Коммиты с изменениями по regex
git log --author="Имя"Коммиты автора
git log --since="2026-01-01" --until="2026-12-31"Коммиты за период

🐛 Bisect (поиск бага)

git bisect startНачать поиск
git bisect badТекущий коммит с багом
git bisect good <commit>Последний хороший коммит
git bisect good / badОтмечать коммиты при поиске
git bisect resetЗавершить поиск

🌳 Worktree

git worktree add ../dir branchСоздать рабочую копию для ветки
git worktree listСписок worktree
git worktree remove ../dirУдалить worktree
git worktree pruneОчистить неактуальные
